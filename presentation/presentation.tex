% --- The pattern ---
\begin{frame}{Why birth regions?}
    \begin{itemize}
        \item Leaders direct resources to where they're \textbf{from}
        \item Hodler \& Raschky (\textit{QJE}, 2014) show this globally in satellite nightlights
        \item The effect is largest in Sub-Saharan Africa
    \end{itemize}
\end{frame}

% --- This project ---
\begin{frame}{This project}
    \textbf{Research questions:}
    \begin{enumerate}
        \item Does \textbf{birth-region favoritism} still show up in nightlights after 2005?
        \item If not, did favoritism shift from \emph{more} growth to \emph{cleaner} growth? (``green favoritism'')
    \end{enumerate}

    \vspace{0.6em}
    \textbf{Our contribution:}
    \begin{itemize}
        \item \textbf{First systematic post-2005 test} of the Hodler--Raschky setup: effect replicates in 1992--2013 ($\beta = 0.013$, $p = 0.030$), but disappears after 2005 in every subsample we tested
        \item \textbf{Introduce \emph{green favoritism}} as a novel composition-shift hypothesis --- and rule it out with satellite NO$_2$ and PM$_{2.5}$
    \end{itemize}
\end{frame}

% --- Data ---
\begin{frame}{Data}
    Global ADM2 district-year panel combining:
    \begin{itemize}
        \item \textbf{PLAD}: leader identity, spell timing, subnational birthplace
        \item \textbf{Nightlights}: DMSP 1992--2013; harmonized DMSP--VIIRS 2005--2019
        \item \textbf{NO$_2$}: ACAG surface NO$_2$, 2005--2019
        \item \textbf{PM$_{2.5}$}: ACAG SatPM, 1998--2024
        \item \textbf{Geography}: GADM ADM2 boundaries
    \end{itemize}

\end{frame}

% --- Empirical strategy ---
\begin{frame}{Empirical strategy: two-way fixed-effects DiD}
    \begin{equation*}
        \log(Y_{ict} + 0.01) = \beta \cdot \text{BirthRegionLeader}_{ict} + \alpha_i + \gamma_{ct} + \varepsilon_{ict}
    \end{equation*}

    \vspace{0.3em}
    \footnotesize\textit{BirthRegionLeader}$_{ict} = 1$ if district $i$ in country $c$ is the birth region of the leader in office in year $t$, $0$ otherwise.
    \normalsize

    \vspace{0.5em}
    \begin{itemize}
        \item ADM2 FE $\alpha_i$, country-year FE $\gamma_{ct}$
        \item $\beta$ compares the leader's birth district to other districts in the same country and year
        \item SEs clustered by leader spell
        \item Outcomes: $\log$ NL, intensity $= \log$ NO$_2 - \log$ NL
    \end{itemize}
\end{frame}

% --- Mechanisms (candidate explanations) ---
\begin{frame}{What could drive the attenuation?}
\small
    \textbf{Four plausible stories:}
    \begin{enumerate}
        \item \textbf{Democratization} --- institutions check leaders' ability to favor home
        \item \textbf{Africa-specific} --- effect was largely an SSA story; maybe survives only there
        \item \textbf{DMSP saturation} --- birth districts too bright to register more growth
        \item \textbf{Composition shift} --- cleaner growth instead of more (``green favoritism'')
    \end{enumerate}

    \vspace{0.5em}
    Each story implies a test:
    \begin{itemize}
        \item Democratization $\Rightarrow$ autocrats only \hfill (\textsc{Result 3})
        \item Africa story $\Rightarrow$ SSA subsample \hfill (\textsc{Result 3})
        \item Saturation $\Rightarrow$ drop bright districts \hfill (\textsc{Result 3})
        \item Green favoritism $\Rightarrow$ pollution outcomes \hfill (\textsc{Result 4})
    \end{itemize}
\end{frame}

% --- Result 4: pollution ---
\begin{frame}{Result 4: no green favoritism}
    \small \textcolor{darkgreen}{If leaders shifted from more growth to cleaner growth, pollution should \emph{fall} in birth regions.}

% --- Takeaways ---
\begin{frame}{Takeaways}
    \begin{enumerate}
        \item The 1992--2013 result \textbf{still replicates}
        \item After 2005 the effect collapses --- and it doesn't come back when we slice the data
        \item None of the four stories holds up:
            \begin{itemize}
                \item Even autocrats don't show it $\Rightarrow$ not democratization
                \item Even Africa doesn't show it post-2005 $\Rightarrow$ not just an Africa story
                \item Even dim districts don't show it $\Rightarrow$ not saturation
                \item Pollution isn't lower in birth regions $\Rightarrow$ not green favoritism
            \end{itemize}
        \item A textbook political-economy result looks much weaker once you push it past 2005
    \end{enumerate}
